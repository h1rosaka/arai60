# STEP 1 自力で解く。
```py
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def sink_here_if_land(i, j):
            if not ( (0 <= i < len(grid)) and (0 <= j < len(grid[0])) ):
                return
            if grid[i][j] == "0":
                return
            # A「今いる場所が1(島)だったら、0にして水に沈める。」
            grid[i][j] = "0"
            # B 周囲4マスに対して、同様にAを再帰的に実施
            sink_here_if_land(i, j + 1)
            sink_here_if_land(i, j - 1)
            sink_here_if_land(i + 1, j)
            sink_here_if_land(i - 1, j)
            
            return 1

        num_islands = 0
        for i in range(len(grid)):
            for j in range(len(grid[0])):
                if sink_here_if_land(i, j) is not None:
                    num_islands += 1
                
        return num_islands

```

```py
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def mark_land_as_visited(row, col):
            if not (0 <= row < height and 0<= col < width):
                return
            if visited[row][col] or grid[row][col] == WATER:
                return
            visited[row][col] = True
            mark_land_as_visited(row - 1, col)
            mark_land_as_visited(row + 1, col)
            mark_land_as_visited(row, col - 1)
            mark_land_as_visited(row, col + 1)

        LAND = "1"
        WATER = "0"
        height, width = len(grid), len(grid[0])

        visited = [ [False] * width for _ in range(height) ]
        num_islands = 0
        for row in range(height):
            for col in range(width):
                if grid[row][col] == LAND and not visited[row][col]:
                    num_islands += 1
                    mark_land_as_visited(row, col)
        return num_islands
```

非破壊BFS
```py
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def mark_land_as_visited(start_row, start_col):
            lands = deque([(start_row, start_col)])
            # 訪問予定リストのlandsに入った時点で、それはもうvisitしたと同義とする
            visited[start_row][start_col] = True
            while lands:
                row, col = lands.popleft()
                dirs = [(1,0), (-1,0), (0,1), (0, -1)]
                for dr, dc in dirs:
                    next_row, next_col = row + dr, col + dc
                    if not (0 <= next_row < height and 0 <= next_col < width):
                        continue
                    if grid[next_row][next_col] == WATER:
                        continue
                    if visited[next_row][next_col]:
                        continue
                    lands.append((next_row, next_col))
                    visited[next_row][next_col] = True

        LAND = "1"
        WATER = "0"
        height = len(grid)
        width = len(grid[0])
        visited = [ [False] * width for _ in range(height) ]
        num_islands = 0
        for row in range(height):
            for col in range(width):
                if grid[row][col] == LAND and not visited[row][col]:
                    num_islands += 1
                    mark_land_as_visited(row, col)

        return num_islands
```

# STEP3　　3回連続でエラーなしで解けるまで解く

```py
class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:

        def mark_visited_and_check_around(row, col):# look aroundの方が自然かな
            # 再帰の終了条件たち
            ## gridの外に出たら終了
            if not (0 <= row < height and 0 <= col < width):
                return
            ## 水だったら終了
            if grid[row][col] == WATER:
                return
            ## すでに訪問済みの土地だったら終了
            if visited_land[row][col]:
                return
            visited_land[row][col] = True
            mark_visited_and_check_around(row + 1, col)
            mark_visited_and_check_around(row - 1, col)
            mark_visited_and_check_around(row, col + 1)
            mark_visited_and_check_around(row, col - 1)

        num_islands = 0
        height = len(grid)
        width = len(grid[0])
        WATER = "0"
        visited_land = [[False] * width for _ in range(height)]
        # 1マスずつ左上から確認
        for row in range(height):
            for col in range(width):
                # 水だったらパス
                if grid[row][col] == WATER:
                    continue
                # 確認済みの陸地だったらパス
                if visited_land[row][col]:
                    continue
                # 島の数を+1して、周囲を確認
                num_islands += 1
                mark_visited_and_check_around(row, col)

        return num_islands
```
