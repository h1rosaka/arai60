# STEP 1 自力で解く。
```py
class Solution:
    def search(self, nums: List[int], target: int) -> int:
        target_index = None
        def bisect_find(nums, x):
            left_index = bisect_left(nums, x)
            if left_index == len(nums):
                return -1
            if nums[left_index] != x:
                return - 1
            return left_index
        # 整列済みの場合
        if nums[0] < nums[-1]:
            return bisect_find(nums, target)
        # 整列済みではない場合、二つの整列ずみに分ける。
        min_num_index = bisect_left(nums, True, key=lambda x: x <= nums[-1])
        ## 右側の場合
        if nums[min_num_index] <= target <= nums[-1]:
            result = bisect_find(nums[min_num_index:], target)
            if result != -1:
                result += min_num_index
            return result
        ## 左側の場合
        return bisect_find(nums[:min_num_index], target)
```
- 最初`if nums[left_index] != x:`の条件を忘れていてWAになった。
- 調べたが、bisect_findみたいな関数はなさそうで、自前で実装が必要そう

```py
class Solution:
    def search(self, nums: List[int], target: int) -> int:
        left = 0
        right = len(nums)
        while right - left > 1:
            mid = (left + right) // 2
            if nums[mid] == target:
                return mid
            # midが右端を含むグループにいる場合
            if nums[mid] <= nums[-1]:
                if nums[mid] <= target <= nums[right - 1]:
                    left = mid
                else:
                    right = mid
            # midが左端を含むグループにいる場合
            else:
                if nums[left] <= target < nums[mid]:
                    right = mid
                else:
                    left = mid
        
        if nums[left] == target:
            return left
        return -1
```

```py
class Solution:
    def search(self, nums: List[int], target: int) -> int:
        def priority(x):
            return (x <= nums[-1], target <= x)
        i = bisect_left(nums, priority(target), key=priority)
        if nums[i] != target:
            return -1
        return i
```
- (True, Flase)のようなboolのタプル同士を比較してbisext_left
- 一つ目のboolが、右側のグループか否か。2つ目のboolがtarget以上か否か。
- 先にどちらのグループに入るのか判定されて、その後グループ内でtarget以上か否かで良いところに入る。
