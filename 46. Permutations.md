    ```py
    class Solution:
        def permute(self, nums: List[int]) -> List[List[int]]:
            result = []

            def helper(current, left_over):
                if not left_over:
                    result.append(current.copy())
                    return

                for i in range(len(left_over)):
                    x = left_over[i]

                    # 樹形図を1段下に状態を進めてhelper実行
                    current.append(x)
                    left_over.pop(i)
                    helper(current, left_over)

                    # 状態を戻す
                    left_over.insert(i, x)
                    current.pop()

            helper([], nums)
            return result
    ```

- 再帰なのでスタックを使ったループに書き直す
    ```py
    """
    [][1,2,3]
    x = 3
    [3], [1,2] → stackへ
    [], [1,2,3]に戻る
    [2] [1,3] → stackへ
    [], [1,2,3]に戻る
    [1] [2,3] →stackへ
    [], [1,2,3]に戻る
    [1] [2,3]がpop
    [1,3][2] →stackへ
    [1,2][3]→ stackへ
    [1,2][3]がpop
    [1,2,3][]→stackへ

    """
    class Solution:
        def permute(self, nums: List[int]) -> List[List[int]]:
            stack = [([], nums)]
            result = []
            while stack:
                partial_permutation, rest = stack.pop()
                if not rest:
                    result.append(partial_permutation)
                    continue
                for i in range(len(rest) - 1, -1, -1):
                    x = rest[i]
                    partial_permutation.append(x)
                    rest.pop(i)
                    stack.append((partial_permutation.copy(), rest.copy())) # copyが必要
                    partial_permutation.pop()
                    rest.insert(i, x)
            return result
    ```
    - Q: なぜこちらはcopyが必要になってしまうのか。
        - A.実行するタイミングに、あるべきpartial_permutationやrestの状態をキープできていないから。
        - なるべく再帰と同じような順番で実行されるように逆順にstackに積んだりしたが、実際には違う順番になってしまっていて、ずれる。（rangeを逆順にしたりして近づけようとしたが、一致はしていない）
            - 再帰ではhelperの処理が実行された直後に、状態を樹形図の一段上のものに戻すが、ここの実装はそうなっていない。
            - →では、helper相当の処理が実行された後に、「状態を一段上のものに戻す」処理もsatckに積んでおけば、同じ動きができるはず↓
            ```py
            class Solution:
                def permute(self, nums: List[int]) -> List[List[int]]:
                    result = []
                    current = []
                    rest = nums
                    # stack frame: ("call", i) or ("resume", i, x)
                    stack = [("call", 0)]
                    while stack:
                        frame = stack.pop()
                        action = frame[0]
                        i = frame[1]
                        if action == "call":
                            if not rest:
                                result.append(current.copy())
                                continue
                            if i < len(rest):
                                x = rest[i]
                                # for ループの次の i(この下でやる「今回のiの処理」の次にやる処理 )
                                stack.append(("call", i + 1))
                                # 今回のiの処理
                                ## 再帰呼び出し戻る用のタスクを事前に積んでおく
                                stack.append(("resume", i, x))
                                ## 再帰呼び出し相当のタスクを積む
                                current.append(x)
                                rest.pop(i)
                                stack.append(("call", 0))

                        else:  # resume
                            x = frame[2]
                            rest.insert(i, x)
                            current.pop()

                    return result
            ```
            - copyなしで書けたけども、copyありのほうが読みやすい気がする。




# STEP3　　3回連続でエラーなしで解けるまで解く

```py
class Solution:
    def permute(self, nums: List[int]) -> List[List[int]]:
        stack = [([], nums)]
        result = []
        while stack:
            partial_permutation, rest = stack.pop()
            if not rest:
                result.append(partial_permutation)
                continue
            for i, num in enumerate(rest):
                new_rest = rest[:i] + rest[i + 1:]
                stack.append((partial_permutation + [num], new_rest))
        return result
```
バックトラッキング版
```py
class Solution:
    def permute(self, nums: List[int]) -> List[List[int]]:
        result = []
        def make_permutation(partial_permutation, rest):
            if not rest:
                result.append(partial_permutation.copy())
            for i, num in enumerate(rest):
                partial_permutation.append(num)
                rest.pop(i)
                make_permutation(partial_permutation, rest)
                partial_permutation.pop()
                rest.insert(i, num)

        make_permutation([], nums)
        return result
```
