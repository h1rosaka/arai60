# STEP 1 自力で解く。
```py
class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        cumulative_sum_to_count = defaultdict(int)
        cumulative_sum_to_count[0] = 1
        num_subarrays_total_k = 0
        cumulative_sum = 0
        # numsを一つずつ見ていく
        for num in nums:
            # ここまでの累積和を計算
            cumulative_sum += num
            # ここで終了するsubarrayを考え、今まで観測した地点の内、合計がkになりうる(都合の良い)開始位置が何個あるか数えて足す
            complement = cumulative_sum - k
            num_subarrays_total_k += cumulative_sum_to_count[complement]
            # 次回以降のために、今回観測した地点の累積和も登録
            cumulative_sum_to_count[cumulative_sum] += 1
        return num_subarrays_total_k

```

    - やりたいこと
        - __init__()をする
            - 親クラスのdictの__init__()を使ってdictクラス作る
            - dict()は、引数に位置引数やキーワード引数を取る様々な初期化方法があるので、args, **kwargsで対応できるようにする

        - 普通のdictとの差分を追加実装する
            - 差分: 存在しないキーだった時に、よしなにvalueをセットしてから返してくれる
            - __missing__をオーバーライドすると、存在しないキーだった時の挙動を指定可能
    - 補足
        - super()は、親クラス差し示したい時に使える便利なもの。
            - >super can be used to refer to parent classes without naming them explicitly, thus making the code more maintainable. 
            - https://docs.python.org/3/library/functions.html#super
        - default_factory は、引数なしで呼び出すと値を返す関数
            - int()は、int(number=0, /)として、デフォルトの0が指定されてる

```py
class mydefaultdict(dict):
  # 辞書として初期化
  def __init__(self, default_factory=None, *args, **kwargs):
    super().__init__(*args, **kwargs)
    self.default_factory = default_factory
  # 存在しないキーだった時に、よしなにセットして返してくれる
  def __missing__(key):
    if self.default_factory is None:
      raise KeyError(key)
    self[key] = self.default_factory()
    return self[key]
```

class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        num_subarrays_total_k = 0
        cumulative_sum = 0
        cumulative_sum_to_location_count = defaultdict(int)
        cumulative_sum_to_location_count[0] = 1
        # numを一つずつ見ていく
        for num in nums:
            # そこまでの累積和を計算し、そこで終了したときに、都合の良い開始地点が何個あるか数えて足す
            cumulative_sum += num
            num_subarrays_total_k += cumulative_sum_to_location_count[cumulative_sum - k]
            # 次回以降のために今回の場所も開始位置として記録
            cumulative_sum_to_location_count[cumulative_sum] += 1
        return num_subarrays_total_k
