class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        WATER = 0
        height = len(grid)
        width = len(grid[0])
        visited = [[False]* width for _ in range(height)]
        def count_area(row, col):
            ## 終了条件たち
            # 外だったら
            if not (0 <= row < height and 0 <= col < width):
                return 0
            # 水だったら
            if grid[row][col] == WATER:
                return 0
            # すでにvisitedだったら
            if visited[row][col]:
                return 0

            visited[row][col] = True
            return (1   + count_area(row + 1, col) 
                        + count_area(row - 1, col)
                        + count_area(row, col + 1)
                        + count_area(row, col - 1))

        max_area = 0
        for row in range(height):
            for col in range(width):
                # 水だったらパス
                if grid[row][col] == WATER:
                    continue
                # 確認済みの土地だったらパス 
                if visited[row][col]:
                    continue
                # 陸だったら島の大きさ測って返す
                area = count_area(row, col)
                max_area = max(max_area, area)
        return max_area

class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        WATER = 0
        height = len(grid)
        width = len(grid[0])
        visited = [[False] * width for _ in range(height)]
        def is_unvisited_land(row, col):
            if not (0 <= row < height and 0 <= col < width):
                return False
            if grid[row][col] == WATER:
                return False
            if visited[row][col]:
                return False
            return True
        
        def count_area(start_row, start_col):
            lands = deque()
            lands.append((start_row, start_col))
            visited[start_row][start_col] = True
            directions = [(1, 0), (-1, 0), (0, 1), (0, -1)]
            area = 0
            while lands:
                row, col = lands.popleft()
                area += 1
                for row_dir, col_dir in directions:
                    new_row , new_col = row + row_dir, col + col_dir
                    if is_unvisited_land(new_row, new_col):
                        lands.append((new_row, new_col))
                        visited[new_row][new_col] = True

            return area
        
        # 左上のマスから1マスずつ見ていく
        max_area = 0
        for row in range(height):
            for col in range(width):
                # 水だったら、パス
                if grid[row][col] == WATER:
                    continue
                # すでに確認済みの陸だったらパス
                if visited[row][col]:
                    continue
                # 陸だったら、大きさチェック　＆　最大値更新
                area = count_area(row, col)
                max_area = max(max_area, area)
        return max_area

Union Find
```py
class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:
        LAND = 1
        WATER = 0
        class UnionFind:
            def __init__(self, n: int):
                self.parents = list(range(n))
                # rootのマスにだけその島の面積を入れる配列。初期状態は全部1（水セルは後で0にする）
                self.sizes = [1] * n

            def find(self, i: int) -> int:
                if self.parents[i] == i:
                    return i
                return self.find(self.parents[i])  # パス圧縮なし

            def union(self, i: int, j: int):
                root_i = self.find(i)
                root_j = self.find(j)
                if root_i == root_j:
                    return
                # root_jを合算後のrootに
                self.parents[root_i] = root_j
                # 合算後のrootのマスに、合計したsizeを格納
                self.sizes[root_j] += self.sizes[root_i]
                # iはルートでは無くなったので保持していたsizeは0にリセット
                self.sizes[root_i] = 0

        if not grid or not grid[0]:
            return 0

        height, width = len(grid), len(grid[0])
        uf = UnionFind(height * width)

        def index(i, j):
            return i * width + j

        # 陸でないところはサイズ0にしておく
        for i in range(height):
            for j in range(width):
                if grid[i][j] == WATER:
                    uf.sizes[index(i, j)] = 0

        for i in range(height):
            for j in range(width):
                if grid[i][j] == LAND:
                    idx = index(i, j)
                    # 右と下のセルが陸だったらunionする
                    for di, dj in [(0,1),(1,0)]:
                        ni, nj = i + di, j + dj
                        if 0 <= ni < height and 0 <= nj < width and grid[ni][nj] == LAND:
                            nidx = index(ni, nj)
                            uf.union(idx, nidx)

        return max(uf.sizes)

```

class Solution:
    def maxAreaOfIsland(self, grid: List[List[int]]) -> int:     
        height = len(grid)
        width = len(grid[0])
        LAND = 1
        seen = [[False] * width for _ in range(height)]
        def traverese_and_compute_area(row, col):
            # termination conditions
            ## outside of grid
            if not (0 <= row < height and 0 <= col < width):
                return 0
            ## not "unseen land"
            if not (grid[row][col] == LAND and not seen[row][col]):
                return 0
            seen[row][col] = True
            return 1 + (traverese_and_compute_area(row + 1, col)
                      + traverese_and_compute_area(row, col + 1)
                      + traverese_and_compute_area(row - 1, col)
                      + traverese_and_compute_area(row, col - 1))
        max_area = 0
        for row in range(height):
            for col in range(width):
                # if current cell is unseen land
                if grid[row][col] == LAND and not seen[row][col]:
                    # compute area and update max
                    area = traverese_and_compute_area(row, col)
                    max_area = max(area, max_area)

        return max_area
