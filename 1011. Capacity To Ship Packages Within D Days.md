```py
class Solution:
    def shipWithinDays(self, weights: List[int], days: int) -> int:
        """
        探し物: days日で運搬可能な容量のうち、もっとも小さい(左)のもの
        L : ここと、ここより左にはない
        R : ここより右にはない
            L < ありうる範囲　<= R  つまり (L, R]
            FFFFFFFTTTTTTTT
        """
        def can_ship_within_the_period(capacity) -> bool:
            i = 0
            for _ in range(days):
                total_weight = 0
                while total_weight + weights[i] <= capacity:
                    total_weight = total_weight + weights[i]
                    i += 1
                    if i == len(weights): # 載せきった。
                        return True
            return False       
        left = 0
        right = sum(weights)
        while right - left > 1:
            mid = (left + right) // 2
            if can_ship_within_the_period(mid):
                right = mid
            else:
                left = mid
        return right
```
