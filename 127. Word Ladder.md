# STEP 1 自力で解く。

```py
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:  
        word_set = set(wordList)
        # endWord入ってるかチェックし、なかったら終わり
        if endWord not in word_set:
            return 0

        word_to_next_possible_words = defaultdict(list)
        for word in [beginWord] + wordList:
            for i in range(len(word)):
                for alphabet in string.ascii_lowercase:
                    new_word = word[:i] + alphabet + word[i + 1:]
                    if new_word in word_set and new_word != word:
                        word_to_next_possible_words[word].append(new_word)

        # 再帰的に探してく。一回使った単語は選ばない。選択肢がなくなるか、end選べたら終わり
            # 引き継ぐ情報は、使用済み単語(これの長さで何回移動したかわかる)
        possible_answers = []
        def keep_finding_next_word(word, used_word_list):
            # 終了条件
            ## endが見つかったら終了
            if endWord in word_to_next_possible_words[word]:
                return possible_answers.append(len(used_word_list) + 2) #はじめと終わりで+2 

            for next_word in word_to_next_possible_words[word]:
                if next_word in used_word_list:
                    continue
                keep_finding_next_word(next_word, used_word_list + [next_word])

            ## nextの単語のなかに移動できるものがなかったら(全部使用済みだったら)終了
            return

        keep_finding_next_word(beginWord, [])
        if not possible_answers:
            return 0
        return min(possible_answers)

```

- 最短の経路だけを知りたかったのに、全部の経路出してから一番短いやつを見つけようとしていたので失敗した
- DFSで解けるならBFSでも解けるのだから、どっちの方が良さそうか先に考えるべきだった。
- あるいは先に自分が紙とペンで解くとしたら、のイメージをしっかりと立てていなかったのが悪いかもしれない。紙に書くならdfsではやらなさそう。
- 今回のように複数の経路を独立に調べる場合は、再帰で引き継ぐ情報に共有のリストを使うとどの時のどの状態かよくわからなくなるので、毎回新しいもの渡した方が良い。(逆にフィボナッチのように一直線の時は、共有で累積和を持ったりして良い)
- 一方、inの演算をするので、リストではなくsetにして、used_words | {next_word}の方が良かった

class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:  
        word_set = set(wordList)
        def next_word_candidates(word):
            candidates = []
            for i in range(len(word)):
                for c in string.ascii_lowercase:
                    new = word[:i] + c + word[i + 1:]
                    if new not in word_set:
                        continue
                    candidates.append(new)
                    word_set.remove(new)

            return candidates

        word_step_q = deque([(beginWord, 1)])
        while word_step_q:
            word, n_steps = word_step_q.popleft()
            if word == endWord:
                return n_steps 
            for word in next_word_candidates(word):
                word_step_q.append((word, n_steps + 1))

        return 0
```
- BFSは近い順に探しているので、一度使った単語はその経路だけでなく今後どんな他の経路でも使われないため、candidates.appendしたらsetから消せる。

class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:  
        word_set = set(wordList)
        def next_candidates(word):
            candidates = []
            for i in range(len(word)):
                for c in string.ascii_lowercase:
                    new = word[:i] + c + word[i + 1:]
                    if new in word_set:
                        candidates.append(new)
                        word_set.remove(new)
            return candidates

        word_step_q = deque([(beginWord, 1)])
        while word_step_q:
            word, step = word_step_q.popleft()
            if word == endWord:
                return step
            for word in next_candidates(word):
                word_step_q.append((word, step + 1))
                
        return 0

```
